2. Sécurité des mots de passe 
2.1 Hashage des mots de passe
	a. La fonction md5() 
	Cote SQL : update user set mdp = md5(mdp); 

	Cote PHP : $mdp = md5($mdp);

	b. La fonction sha1()
	Cote SQL : 
		update user set mdp = "123" where iduser = 1; 
		update user set mdp = "456" where iduser = 2; 
		update user set mdp = sha1(mdp); 

	Cote PHP : 
		$mdp = sha1($mdp); 

	c. Renforcement avec le grain sel 
	Le grain sel est un très grand nombre qui s'ajoute au mdp auxquels est applique le sha1. 

	Coté SQL : 
		create table grainSel (
		nb varchar(200) not null primary key
		);
		insert into grainSel values("1234567890"); 

		update user set mdp = "123" where iduser = 1; 
		update user set mdp = "456" where iduser = 2;

		set @gsel = "";
		select @gsel; 
		select nb into @gsel from grainsel ;
		update user set mdp = sha1(concat(mdp, @gsel ));

	Cote PHP : 
		methode dans le modele pour récupérer le grainSel 

		on applique le sha1 à la concaténation 
		$mdp = sha1($mdp.$gsel);
